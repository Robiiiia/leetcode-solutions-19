linked list cycle

public class Solution {

    public boolean hasCycle(ListNode head) {

        // Если список пустой, значит, цикла нет.
        if (head == null) {
            return false;
        }

        // Медленный указатель начинает с первого узла.
        ListNode slow = head;

        // Быстрый указатель тоже начинает с первого узла.
        ListNode fast = head;

        // Пока быстрый указатель и следующий узел существуют,
        // продолжаем проверять список.
        while (fast != null && fast.next != null) {

            // Медленный указатель перемещается на один узел вперёд.
            slow = slow.next;

            // Быстрый указатель перемещается на два узла вперёд.
            fast = fast.next.next;

            // Если указатели встретились, значит, цикл существует.
            if (slow == fast) {
                return true;
            }
        }

        // Если быстрый указатель дошёл до конца,
        // значит, цикла нет.
        return false;
    }
}
